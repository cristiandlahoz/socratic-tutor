        #include <stdio.h>

/* Función para validar que un número sea mayor que 0 */
float validarMayorQueCero(float valor) {
    while (valor <= 0) {
        printf("Valor invalido. Introduzca un numero mayor que 0: ");
        scanf("%f", &valor);
    }
    return valor;
}

int main() {
    int n, contador = 1;
    int id;
    float practica, parcial, proyecto, notaFinal;

    printf("Introduzca la cantidad de estudiantes: ");
    scanf("%d", &n);

    n = (int)validarMayorQueCero((float)n);

    while (contador <= n) {

        printf("\n=============================\n");
        printf("Estudiante #%d\n", contador);
        printf("=============================\n");

        printf("Introduzca ID: ");
        scanf("%d", &id);
        id = (int)validarMayorQueCero((float)id);

        printf("Introduzca nota practica: ");
        scanf("%f", &practica);
        practica = validarMayorQueCero(practica);

        printf("Introduzca nota parcial: ");
        scanf("%f", &parcial);
        parcial = validarMayorQueCero(parcial);

        printf("Introduzca nota proyecto: ");
        scanf("%f", &proyecto);
        proyecto = validarMayorQueCero(proyecto);

        notaFinal = (practica * 0.30f) +
                    (parcial * 0.30f) +
                    (proyecto * 0.40f);

        contador = contador + 1;
    }

    return 0;
}
