Así resolvemos el problema utilizando while anidados.

#include <stdio.h>

int main(void) {

    /* =========================
       NIVEL 1: ESTUDIANTES
       =========================
       Este while recorre cada estudiante.
       Por cada estudiante, dentro, se ejecutará el NIVEL 2 (materias).
    */

    int nEstudiantes;
    int contadorEst;

    /* =========================
       ACUMULADOR GENERAL
       =========================
       Debe ir FUERA del while de estudiantes porque
       debe acumular lo de TODOS los estudiantes.
    */
    float totalGanadoUniversidad = 0;

    printf("Introduzca la cantidad de estudiantes a evaluar: ");
    scanf("%d", &nEstudiantes);

    /* Contador del NIVEL 1 */
    contadorEst = 1;

    while (contadorEst <= nEstudiantes) {

        printf("\n==============================\n");
        printf("  Estudiante #%d\n", contadorEst);
        printf("==============================\n");

        /* =========================
           DENTRO DE CADA ESTUDIANTE
           =========================
           Vamos a pedir cuántas materias tiene este estudiante.
        */
        int nMaterias;
        printf("Introduzca la cantidad de materias del estudiante #%d: ", contadorEst);
        scanf("%d", &nMaterias);

        /* =========================
           ACUMULADOR POR ESTUDIANTE
           =========================
           Debe reiniciarse en 0 por cada estudiante.
           Por eso va DENTRO del while de estudiantes.
        */
        float totalEstudiante = 0;

        /* =========================
           NIVEL 2: MATERIAS
           =========================
           Este while recorre cada materia de ESTE estudiante.
        */
        int contadorMat = 1;

        while (contadorMat <= nMaterias) {

            printf("\n--- Materia #%d ---\n", contadorMat);

            /* Datos necesarios por materia */
            int creditos;
            float precioCredito;

            printf("Introduzca la cantidad de creditos: ");
            scanf("%d", &creditos);

            printf("Introduzca el precio por credito: ");
            scanf("%f", &precioCredito);

            /* Costo de esta materia */
            float costoMateria = creditos * precioCredito;

            printf("Costo de la materia #%d: %.2f\n", contadorMat, costoMateria);

            /* =========================
               INVARIANTE DEL NIVEL 2
               =========================
               En cada vuelta, totalEstudiante va guardando
               la suma parcial de las materias registradas.
            */
            totalEstudiante = totalEstudiante + costoMateria;

            /* Avanzamos a la siguiente materia */
            contadorMat = contadorMat + 1;
        }

        /* Aquí termina NIVEL 2.
           totalEstudiante ya es el total final del estudiante. */
        printf("\nTOTAL PAGADO por el estudiante #%d: %.2f\n", contadorEst, totalEstudiante);

        /* =========================
           ACUMULACIÓN GENERAL
           =========================
           Se suma el total del estudiante al total de la universidad.
        */
        totalGanadoUniversidad = totalGanadoUniversidad + totalEstudiante;

        /* Avanzamos al siguiente estudiante (NIVEL 1) */
        contadorEst = contadorEst + 1;
    }

    /* Al terminar el NIVEL 1, ya tenemos el total general */
    printf("\n=====================================\n");
    printf("TOTAL GANADO POR LA UNIVERSIDAD: %.2f\n", totalGanadoUniversidad);
    printf("=====================================\n");

    return 0;
}
